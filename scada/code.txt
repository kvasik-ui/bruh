Вариант 27. SCADA для энергетики (C++, IEC 61850, DNP3, визуализация)

# IDE Qt Creator и инструменты сборки
sudo apt install qtcreator g++ cmake libssl-dev git -y

# libiec61850: клонирование и сборка
git clone https://github.com/mz-automation/libiec61850.git
cd libiec61850 && mkdir build && cd build
cmake .. && make -j$(nproc)
cd ~

# OpenDNP3: клонирование и сборка
git clone --recursive https://github.com/dnp3/opendnp3.git
cd opendnp3 && mkdir build && cd build
cmake .. && make -j$(nproc)
cd ~

# Grafana через Docker (в apt её нет)
sudo apt install docker.io -y
sudo systemctl enable --now docker
sudo docker run -d --name grafana -p 127.0.0.1:3000:3000 grafana/grafana

# Проверка: Grafana отвечает на порту 3000
curl -I http://localhost:3000
