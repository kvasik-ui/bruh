Вариант 19. Многопользовательские онлайн-игры (серверная логика C++/Python, нагрузка)

# Компилятор C++ и библиотеки для сетевого сервера
sudo apt install g++ cmake libboost-all-dev -y

# Балансировка, мониторинг и базовая защита
sudo apt install haproxy prometheus fail2ban ufw -y

# Locust в отдельном окружении
sudo apt install python3-pip python3-venv -y
python3 -m venv ~/game-env && source ~/game-env/bin/activate
pip install locust
locust --version
deactivate

# JMeter из apt (версия устаревшая; актуальную — с jmeter.apache.org)
sudo apt install jmeter -y

# Проверка: версии и состояние служб
g++ --version && haproxy -v | head -1
systemctl is-active prometheus fail2ban
