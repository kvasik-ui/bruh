Вариант 25. Цифровые фабрики: промышленные протоколы (C#)

# IDE Rider (snap)
sudo snap install rider --classic

# .NET SDK 8
sudo apt install dotnet-sdk-8.0 -y

# Клиентский проект и библиотека OPC UA
dotnet new console -n FactoryOPC && cd FactoryOPC
dotnet add package OPCFoundation.NetStandard.Opc.Ua
dotnet build
cd ~

# Тестовый OPC UA сервер (Python, asyncua), чтобы проверить обмен данными
sudo apt install python3-pip python3-venv -y
python3 -m venv ~/opcua-env && source ~/opcua-env/bin/activate
pip install asyncua
deactivate

# Брокер MQTT для телеметрии
sudo apt install mosquitto mosquitto-clients -y
sudo systemctl enable --now mosquitto
