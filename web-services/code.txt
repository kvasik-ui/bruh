Вариант 1. Высоконагруженные веб-сервисы (Python, JS, Django, Node.js)

# IDE для Python и Django (PyCharm Community, snap). Проверить имя: snap find pycharm
sudo snap install pycharm-community --classic

# Редактор для JavaScript и Node.js (VS Code, snap)
sudo snap install code --classic

# Node.js и npm из репозитория Ubuntu
sudo apt install nodejs npm -y

# Python: виртуальное окружение и Django
sudo apt install python3-pip python3-venv -y
python3 -m venv ~/web-env && source ~/web-env/bin/activate
pip install django
django-admin --version
deactivate

# Git-клиент GitKraken (deb)
wget https://release.gitkraken.com/linux/gitkraken-amd64.deb
sudo dpkg -i gitkraken-amd64.deb || sudo apt -f install -y

# Сервер баз данных PostgreSQL
sudo apt install postgresql postgresql-contrib -y

# Инструмент тестирования API (snap)
sudo snap install postman

# Проверка версий
python3 --version && node --version && psql --version
