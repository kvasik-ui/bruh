Вариант 28. Биометрические системы идентификации (Python, компьютерное зрение, камеры)

# Инструменты сборки для dlib, OpenCV и NumPy из apt
sudo apt install build-essential cmake libboost-all-dev python3-opencv python3-numpy python3-venv python3-pip -y

# Окружение с доступом к системным пакетам
python3 -m venv --system-site-packages ~/bio-env && source ~/bio-env/bin/activate

# dlib компилируется из исходников (может занять несколько минут)
pip install dlib face_recognition onvif-zeep

# Проверка: OpenCV и face_recognition импортируются в одном окружении
python -c "import cv2, face_recognition; print('OK')"
deactivate
